- [ ] **Step 3: Write the codegen script**

`scripts/generate-svg-assets.mjs`:

```js
// Generates TS modules from the SVG assets so the engine core needs no
// bundler-specific ?raw imports (works under ng-packagr, Vitest and the CLI).
// Output is committed; re-run with `npm run gen:assets` after editing an SVG.
import { readFileSync, writeFileSync, readdirSync } from "node:fs";
import { resolve, dirname, basename } from "node:path";
import { fileURLToPath } from "node:url";

const here = dirname(fileURLToPath(import.meta.url));
const core = resolve(here, "../projects/angular-advanced-odontogram/src/lib/core");
const HEADER = "// AUTO-GENERATED by scripts/generate-svg-assets.mjs — do not edit.\n" +
  "// Re-run `npm run gen:assets` after changing an asset SVG.\n\n";

// 11.svg -> tooth11Svg, 14_occl.svg -> tooth14OcclSvg (must match the import
// names previously used by the ?raw imports in odontogram.ts).
const toothExport = (file) => {
  const stem = basename(file, ".svg");
  const [no, variant] = stem.split("_");
  return `tooth${no}${variant ? variant[0].toUpperCase() + variant.slice(1) : ""}Svg`;
};
// icon_8.svg -> icon8Svg, icon_gum.svg -> iconGumSvg (match App.tsx names;
// a digit stem like "8" is unchanged by toUpperCase, so no special case).
const iconExport = (file) => {
  const stem = basename(file, ".svg").replace(/^icon_/, "");
  return `icon${stem[0].toUpperCase() + stem.slice(1)}Svg`;
};

function emit(dir, outFile, exportName, extra = "") {
  const src = resolve(core, "assets", dir);
  const files = readdirSync(src).filter((f) => f.endsWith(".svg")).sort();
  let out = HEADER;
  for (const f of files) {
    if (extra && f === "icon_no_selection.svg") continue; // emitted as data URI below
    const markup = readFileSync(resolve(src, f), "utf8");
    out += `export const ${exportName(f)} = ${JSON.stringify(markup)};\n`;
  }
  out += extra;
  writeFileSync(resolve(core, "generated", outFile), out);
  console.log(`wrote generated/${outFile} (${files.length} assets)`);
}

import { mkdirSync } from "node:fs";
mkdirSync(resolve(core, "generated"), { recursive: true });
emit("teeth-svgs", "teeth-svgs.ts", toothExport);
const noSel = readFileSync(resolve(core, "assets/icon-svgs/icon_no_selection.svg"), "utf8");
emit("icon-svgs", "icon-svgs.ts", iconExport,
  `export const iconNoSelectionUrl = "data:image/svg+xml," + encodeURIComponent(${JSON.stringify(noSel)});\n`);
```

The **generated export names are the contract** (they must match the Interfaces block above); Step 4 verifies them mechanically.
